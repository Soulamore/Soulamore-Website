        // --- 1. STATE & API ---
        let feedData = [];

        async function initFeed() {
            const feedContainer = document.getElementById('postsFeed');
            feedContainer.innerHTML = '<div style="padding:20px; text-align:center; color:var(--text-secondary);">Loading community posts...</div>';

            try {
                // Fetch from Service Layer
                feedData = await window.soulamoreApi.getFeed();
                renderFeed();
            } catch (err) {
                console.error("Error loading feed:", err);
                feedContainer.innerHTML = '<div style="color:red; text-align:center;">Failed to load posts. Please try again.</div>';
            }
        }

        // --- 2. RENDER FEED ---
        const feedContainer = document.getElementById('postsFeed');
        let currentFilter = 'All';

        function filterFeed(category, el) {
            document.querySelectorAll('.filter-pill').forEach(p => p.classList.remove('active'));
            el.classList.add('active');
            currentFilter = category;
            renderFeed();
        }

        function renderFeed() {
            const filteredData = feedData.filter(post => {
                if (currentFilter === 'All') return true;
                if (currentFilter === 'Instagram') return post.source === 'instagram';
                if (currentFilter === 'Confessions') return post.source === 'confession';
                return post.tag === currentFilter;
            });

            if (filteredData.length === 0) {
                feedContainer.innerHTML = '<div style="padding:40px; text-align:center; opacity:0.6;">No posts found in this category yet.</div>';
                return;
            }

            feedContainer.innerHTML = filteredData.map(post => {
                let badge = '';
                let actionButton = '';

                if (post.source === 'instagram') {
                    badge = `<span class="tag-pill" style="background:rgba(193, 53, 132, 0.2); color:#C13584; margin-left:10px;"><i class="fab fa-instagram"></i> From Instagram</span>`;
                    actionButton = `<div class="action-btn" onclick="event.stopPropagation(); window.open('${post.instagram_post_url}')" style="color:#C13584;"><i class="fas fa-external-link-alt"></i> View on Insta</div>`;
                } else if (post.source === 'confession') {
                    badge = `<span class="tag-pill" style="background:rgba(78, 205, 196, 0.2); color:white; margin-left:10px;"><i class="fas fa-mask"></i> Curated Confession</span>`;
                    actionButton = `<div class="action-btn" onclick="event.stopPropagation(); window.open('${post.instagram_reel_url}')" style="color:var(--teal-glow);"><i class="fas fa-play-circle"></i> View Reel</div>`;
                }

                return `
                <div class="post-card fade-in" onclick="openPost(${post.id})">
                    <div class="post-meta">
                        <span style="font-weight:700; color:var(--starlight);">${post.author}</span>
                        <span>• ${post.time}</span>
                        <span class="tag-pill">${post.tag}</span>
                        ${badge}
                    </div>
                    <div class="post-title">${post.title}</div>
                    <div class="post-preview">${post.content}</div>
                    <div class="post-actions">
                        <div class="action-btn"><i class="fas fa-heart"></i> ${post.upvotes} Support</div>
                        <div class="action-btn"><i class="fas fa-comment"></i> ${post.source === 'confession' ? 'View Comments' : post.comments + ' Comments'}</div>
                        <div class="action-btn"><i class="fas fa-share"></i> Share</div>
                        ${actionButton}
                    </div>
                </div>
            `}).join('');
        }
